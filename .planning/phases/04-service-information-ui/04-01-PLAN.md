  <behavior>
    getOpenStatus(aukioloajat, now) — returns OpenStatus:

    Case 1: null/undefined input
      - getOpenStatus(null) → { status: 'no-data' }
      - getOpenStatus(undefined) → { status: 'no-data' }

    Case 2: aukioloajat present, today's key absent (day not in object)
      - Input: { monday: { open: '09:00', close: '21:00' } }, now = Sunday
      - → { status: 'closed', hours: null }
      (other days have data, so not 'no-data'; today just has no entry)

    Case 3: aukioloajat present, today has empty strings
      - Input: { sunday: { open: '', close: '' } }, now = Sunday
      - → { status: 'closed', hours: null }
      (empty string guard: entry.open is falsy)

    Case 4: now is within range (same day)
      - Input: { monday: { open: '09:00', close: '21:00' } }, now = Monday 10:00
      - → { status: 'open', hours: '09:00–21:00' }

    Case 5: now is outside range (same day)
      - Input: { monday: { open: '09:00', close: '21:00' } }, now = Monday 22:00
      - → { status: 'closed', hours: '09:00–21:00' }

    Case 6: after-midnight close (closeMins < openMins)
      - Input: { friday: { open: '22:00', close: '02:00' } }, now = Friday 23:00
      - → { status: 'open', hours: '22:00–02:00' }
      - Input: same, now = Friday 01:00 (early morning, still "open" from night before)
      — WAIT: this is Saturday getDay(), not Friday. The test must use Friday 23:00 only.
      - Input: same, now = Friday 08:00 (before opening)
      - → { status: 'closed', hours: '22:00–02:00' }

    Case 7: all-zero object (no day with truthy open/close)
      - Input: { monday: { open: '', close: '' }, tuesday: { open: '', close: '' } }
      - → { status: 'no-data' } (hasAnyData check returns false)

    formatGroupedHours(aukioloajat) — returns HourGroup[]:

    Case 8: null/undefined
      - → []

    Case 9: Ma–Pe same hours, La different, Su closed
      - Input: monday–friday all '06:00–22:00', saturday '09:00–18:00', sunday absent
      - → [
          { key: 'monday-tuesday-wednesday-thursday-friday', label: 'Ma–Pe', hours: '06:00–22:00', dayKeys: ['monday','tuesday','wednesday','thursday','friday'] },
          { key: 'saturday', label: 'La', hours: '09:00–18:00', dayKeys: ['saturday'] },
          { key: 'sunday', label: 'Su', hours: 'suljettu', dayKeys: ['sunday'] }
        ]

    Case 10: All days closed (no data)
      - Input: {}
      - → 7 groups each with hours: 'suljettu'

    Case 11: Single day different from neighbors
      - Input: monday '09:00–21:00', tuesday '07:00–23:00', wednesday '09:00–21:00'
      - No grouping of monday+wednesday (tuesday breaks the run)
      - → 3 separate groups for those days (remaining days grouped by their hours)
  </behavior>

    Step 3 — lib/aukiolo.ts (GREEN):

    Export types:
      - OpenStatus: discriminated union with status 'open' | 'closed' | 'no-data'
        - open: { status: 'open'; hours: string }
        - closed: { status: 'closed'; hours: string | null }
        - no-data: { status: 'no-data' }
      - HourGroup: { key: string; label: string; hours: string; dayKeys: string[] }

    getOpenStatus implementation:
      - DAY_KEYS constant: ['sunday','monday','tuesday','wednesday','thursday','friday','saturday']
      - Guard: if (!aukioloajat) return { status: 'no-data' }
      - dayKey = DAY_KEYS[now.getDay()]
      - entry = aukioloajat[dayKey]
      - If !entry || !entry.open || !entry.close:
          hasAnyData = Object.values(aukioloajat).some(v => v.open && v.close)
          if (!hasAnyData) return { status: 'no-data' }
          return { status: 'closed', hours: null }
      - nowMins = now.getHours() * 60 + now.getMinutes()
      - openMins = parseInt(entry.open.split(':')[0]) * 60 + parseInt(entry.open.split(':')[1])
      - closeMins = parseInt(entry.close.split(':')[0]) * 60 + parseInt(entry.close.split(':')[1])
      - After-midnight: if closeMins < openMins → isOpen = nowMins >= openMins || nowMins < closeMins
      - Same-day: isOpen = nowMins >= openMins && nowMins < closeMins
      - rangeStr = entry.open + '–' + entry.close  (en-dash U+2013)
      - Return { status: isOpen ? 'open' : 'closed', hours: rangeStr }

    formatGroupedHours implementation:
      - ORDERED_DAYS: ['monday','tuesday','wednesday','thursday','friday','saturday','sunday']
      - FI_ABBR map: monday→'Ma', tuesday→'Ti', wednesday→'Ke', thursday→'To', friday→'Pe', saturday→'La', sunday→'Su'
      - if (!aukioloajat) return []
      - Index-walk while loop:
          i = 0, groups = []
          while i < 7:
            dayKey = ORDERED_DAYS[i]
            entry = aukioloajat[dayKey]
            hours = (entry?.open && entry?.close) ? entry.open + '–' + entry.close : 'suljettu'
            j = i + 1
            while j < 7:
              nextKey = ORDERED_DAYS[j]
              nextEntry = aukioloajat[nextKey]
              nextHours = (nextEntry?.open && nextEntry?.close) ? nextEntry.open + '–' + nextEntry.close : 'suljettu'
              if nextHours !== hours: break
              j++
            span = ORDERED_DAYS.slice(i, j)
            label = span.length === 1 ? FI_ABBR[span[0]] : FI_ABBR[span[0]] + '–' + FI_ABBR[span[span.length-1]]
            groups.push({ key: span.join('-'), label, hours, dayKeys: [...span] })
            i = j
          return groups

    NOTE: Do NOT use new Date('HH:MM') for time parsing — it is unreliable. Always use split(':') and parseInt.
    NOTE: The en-dash in rangeStr is U+2013 (–), not a hyphen (-).
  </implementation>
