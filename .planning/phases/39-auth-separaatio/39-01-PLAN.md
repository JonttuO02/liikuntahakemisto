import { createServerClient } from '@supabase/ssr'
import type { ReadonlyRequestCookies } from 'next/dist/server/web/spec-extension/adapters/request-cookies'

export function createServerSupabase(cookieStore: ReadonlyRequestCookies) {
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() { return cookieStore.getAll() },
        setAll() {}, // server components cannot set cookies — middleware handles refresh
      },
    }
  )
}
```

import { createClient, type SupabaseClient } from '@supabase/supabase-js'

let _browserClient: BrowserClient | undefined

export function createBrowserSupabase(): BrowserClient {
  if (!_browserClient) {
    _browserClient = createClient<any>(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    )
    _browserClient.auth.onAuthStateChange(...)
  }
  return _browserClient
}
```

From @supabase/ssr v0.10.3 source (cookie namespace mechanism):
- createServerClient accepts options.cookieOptions.name — when set, it becomes storageKey in auth-js
- createBrowserClient accepts options.cookieOptions.name — same mechanism
- Setting cookieOptions.name = 'sb-biz' produces cookies named sb-biz-auth-token, sb-biz-auth-token.0, etc.
- The @supabase/ssr createBrowserClient has its own internal singleton (cachedBrowserClient) BUT it is keyed to a single instance — since we need an independent business singleton, we wrap it in a module-level variable ourselves (do NOT use createBrowserClient's built-in singleton; use our own _bizBrowserClient like supabaseSSR.ts does with createClient).
