<!-- createServerSupabase usage (from lib/supabaseSSR.ts): -->
import { cookies } from 'next/headers'
import { createServerSupabase } from '@/lib/supabaseSSR'

export default async function BusinessPage() {
  const cookieStore = cookies()
  const supabase = createServerSupabase(cookieStore)
  const { data: { user } } = await supabase.auth.getUser()
  // user is null if not authenticated
  
  if (user) {
    // Check business_paikka_links
    const { data: links } = await supabase
      .from('business_paikka_links')
      .select('paikka_id, liikuntapaikat(nimi)')
      .eq('business_account_id', user.id)
      .limit(1)
    // links === null or links.length === 0 → Path A
    // links.length > 0 → Path B
  }
  // No user → Path A (consistent with Phase 32 D-08: no middleware guard yet)
}
