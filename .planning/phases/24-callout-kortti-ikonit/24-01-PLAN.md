```typescript
function CalloutCard({ p }: { p: Liikuntapaikka & { latitude: number; longitude: number } }) {
  const ref = useRef<HTMLDivElement>(null)
  const [clipPath, setClipPath] = useState('')

  useLayoutEffect(() => {
    const el = ref.current
    if (!el) return
    const compute = () => {
      const h = el.offsetHeight - 11
      setClipPath(
        `M 10,0 L 120,0 Q 130,0 130,10 L 130,${h - 10} Q 130,${h} 120,${h} L 75,${h} L 65,${h + 11} L 55,${h} L 10,${h} Q 0,${h} 0,${h - 10} L 0,10 Q 0,0 10,0 Z`
      )
    }
    const obs = new ResizeObserver(compute)
    obs.observe(el)
    compute()
    return () => obs.disconnect()
  }, [])

  const hinta = p.hinta_kuvaus || hintateksti(p.hinta_min, p.hinta_max)

  return (
    <div ref={ref} className="glass px-2.5 py-2 flex flex-col gap-1 cursor-pointer"
      style={{ width: 130, paddingBottom: 11, clipPath: clipPath ? `path('${clipPath}')` : undefined, borderRadius: clipPath ? 0 : 10 }}>
      <span className="inline-flex items-center text-[10px] font-bold px-1.5 py-0.5 rounded-full text-white truncate"
        style={{ backgroundColor: lajiKonfig[p.laji]?.color ?? '#6b7280' }}>
        {lajiKonfig[p.laji]?.label ?? p.laji}
      </span>
      <span className="font-bold text-sm text-[#111111] truncate leading-tight">{p.nimi}</span>
      {hinta && <span className="text-[10px] text-[rgba(17,17,17,0.55)] truncate">{hinta}</span>}
    </div>
  )
}
```
